enable
configure terminal
hostname EdgeRouter-Nhom8

! 1. Inside interface (Link to CoreSW-Nhom8)
interface GigabitEthernet0/0
 description Link to CoreSW-Nhom8 (Fa0/24)
 ip address 10.0.0.1 255.255.255.252
 ip nat inside
 no shutdown
 exit
!
! 2. Outside interface (Link to ISP-Internet)
interface GigabitEthernet0/1
 description Link to ISP-Internet (Gi0/0)
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no shutdown
 exit
!
! 3. Default Route to ISP
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
! 4. Static Return Routes to internal VLANs via CoreSW-Nhom8
ip route 192.168.10.0 255.255.255.0 10.0.0.2
ip route 192.168.20.0 255.255.255.0 10.0.0.2
ip route 192.168.30.0 255.255.255.0 10.0.0.2
ip route 192.168.40.0 255.255.255.0 10.0.0.2
ip route 192.168.50.0 255.255.255.0 10.0.0.2
!
! 5. NAT/PAT Overload for valid VLANs & CoreSW Uplink (Exclude VLAN 99 Attacker)
access-list 1 permit 10.0.0.0 0.0.0.3
access-list 1 permit 192.168.10.0 0.0.0.255
access-list 1 permit 192.168.20.0 0.0.0.255
access-list 1 permit 192.168.30.0 0.0.0.255
access-list 1 permit 192.168.40.0 0.0.0.255
access-list 1 permit 192.168.50.0 0.0.0.255
ip nat inside source list 1 interface GigabitEthernet0/1 overload
!
end
write memory
